  res.json({

  res.json({

    timestamp: new Date().toISOString()
  });
});

  const { email } = req.body;
  // demo only
  res.json({
    token: "demo-token",
    user: { email }
  });
});

app.post("/api/auth/refresh", (_req, res) => {
  res.json({ token: "demo-refreshed-token" });
});

  console.log(`Nexus One backend running on http://localhost:${port}`);

export const NexusSymbol: React.FC = () => {
  const { symbol } = nexusConfig;
  const { primaryColor, accentColor, strokeWidth } = symbol.style;
  const radius = symbol.geometry.centerNode.radius;
  const connections = symbol.geometry.connections;

  const size = 200;
  const center = size / 2;

  const dirToOffset = (direction: string, length: number) => {
    switch (direction) {
      case "north":
        return { x: 0, y: -length };
      case "south":
        return { x: 0, y: length };
      case "east":
        return { x: length, y: 0 };
      case "west":
        return { x: -length, y: 0 };
      default:
        return { x: 0, y: 0 };
    }
  };

  return (
    <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`}>
      {/* background */}
      <rect width={size} height={size} fill="#0D1117" rx="16" />
      {/* connections */}
      {connections.map((c: any, i: number) => {
        const offset = dirToOffset(c.direction, c.length / 2);
        return (
          <line
            key={i}
            x1={center}
            y1={center}
            x2={center + offset.x}
            y2={center + offset.y}
            stroke={accentColor}
            strokeWidth={strokeWidth}
            strokeLinecap="round"
          />
        );
      })}
      {/* core node */}
      <circle
        cx={center}
        cy={center}
        r={radius}
        fill={primaryColor}
        stroke={accentColor}
        strokeWidth={strokeWidth}
      />
    </svg>
  );
};
return res.json();
}
